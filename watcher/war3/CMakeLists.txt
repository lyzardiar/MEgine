cmake_minimum_required(VERSION 3.24)
project(War3 VERSION 0.1.0 LANGUAGES C CXX)
if(NOT MSVC OR NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "War3 requires MSVC targeting Win32 (x86). Use the windows-x86 preset.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(STORM_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(STORM_USE_BUNDLED_LIBRARIES ON CACHE BOOL "" FORCE)
set(STORM_UNICODE ON CACHE BOOL "" FORCE)
set(STORM_SKIP_INSTALL ON CACHE BOOL "" FORCE)
add_subdirectory(third_party/stormlib/source EXCLUDE_FROM_ALL)
add_library(war3_core STATIC src/platform.cpp src/archive.cpp src/image.cpp)
target_compile_definitions(war3_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(war3_core PRIVATE /W4 /permissive- /utf-8)
target_link_libraries(war3_core PUBLIC StormLib::storm shell32 bcrypt version)
target_include_directories(war3_core PUBLIC src)
add_executable(war3 WIN32 src/main.cpp src/renderer.cpp src/media.cpp)
target_compile_options(war3 PRIVATE /W4 /permissive- /utf-8)
target_link_libraries(war3 PRIVATE war3_core d3d9 gdi32)
set_target_properties(war3 PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
